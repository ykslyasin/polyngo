import React, { useState } from "react";
import { Box, Button, TextField, Grid } from "@mui/material";
import { Typography } from "@mui/joy";

const KomutUretColumn = ({ tools }) => {
  const [selectedTool, setSelectedTool] = useState(null);
  const [selectedFeature, setSelectedFeature] = useState(null);
  const [inputValues, setInputValues] = useState({});
  const [generatedCommand, setGeneratedCommand] = useState("");

  const handleToolSelect = (tool) => {
    if (selectedTool === tool) {
      // Eğer aynı araç seçilmişse, seçimi temizle
      setSelectedTool(null);
      setSelectedFeature(null);
      setInputValues({});
      setGeneratedCommand("");
    } else {
      // Yeni bir araç seçilmişse, onu ayarla
      setSelectedTool(tool);
      setSelectedFeature(null);
      setInputValues({});
      setGeneratedCommand("");
    }
  };
  

  const handleFeatureSelect = (feature) => {
    if (selectedFeature === feature) {
      // Eğer aynı özellik seçilmişse, seçimi temizle
      setSelectedFeature(null);
      setInputValues({});
      setGeneratedCommand("");
    } else {
      // Yeni bir özellik seçilmişse, onu ayarla
      setSelectedFeature(feature);
      setInputValues({});
      setGeneratedCommand("");
    }
  };


    const [isClicked, setIsClicked] = useState(false);
  
    const handleClick = () => {
      setIsClicked(!isClicked);
    };
  

  const handleInputChange = (key, value) => {
    setInputValues((prev) => ({
      ...prev,
      [key]: value,
    }));
  };

  const generateCommand = () => {
    if (!selectedFeature) return;
    const command = selectedFeature.template.replace(/\{(\w+)\}/g, (_, key) => inputValues[key] || "");
    setGeneratedCommand(command);
  };

  

  return (
    


    <Box sx={{ padding: 3 }}>

<Button
      variant="text"
      sx={{
        color: isClicked ? 'rgb(255,215,0)' : 'rgb(169,169,169)', // Altın sarısı ya da gri
      }}
      onClick={handleClick}
    >
      ★
    </Button>

      <Typography variant="h6" sx={{ color: "rgb(210,210,210)" }}></Typography>
      {/* Araçlar için yan yana düzen */}
      <Box
        sx={{
          display: "flex",
          flexWrap: "wrap",
          gap: 1,
          marginBottom: 2,
        }}
      >
        {tools.map((tool) => (
          <Button
            key={tool.name}
            variant="contained"
            onClick={() => handleToolSelect(tool)}
            sx={{ backgroundColor: "rgb(255,0,71)" }}
          >
            {tool.name}
          </Button>
        ))}
      </Box>

      {/* Seçilen aracın özelliklerini göster */}
      {selectedTool && (
        <Box>
          <Typography variant="h6" sx={{ color: "rgb(210,210,210)" }}>
            Özellikler ({selectedTool.name})
          </Typography>
          <Box
            sx={{
              display: "flex",
              flexWrap: "wrap",
              gap: 1,
              marginBottom: 2,
            }}
          >
            {selectedTool.features.map((feature) => (
              <Button
                key={feature.name}
                variant="outlined"
                onClick={() => handleFeatureSelect(feature)}
                sx={{
                  color: "rgb(255,0,71)",
                  borderColor: "rgb(255,0,71)",
                }}
              >
                {feature.name}
              </Button>
            ))}
          </Box>
        </Box>
      )}

      {/* Seçilen özelliğin parametrelerini göster */}
      {selectedFeature && (
        <Box>
          <Typography variant="h6" sx={{ color: "rgb(210,210,210)" }}>
            Parametreler ({selectedFeature.name})
          </Typography>
          {selectedFeature.inputs.map((input) => (
            <TextField
  key={input.key}
  label={input.label}
  variant="outlined"
  fullWidth
  margin="normal"
  onChange={(e) => handleInputChange(input.key, e.target.value)}
  InputProps={{
    style: {
      color: "rgb(255,0,71)", // İç yazı rengini kırmızı yapar
    },
  }}
  InputLabelProps={{
    style: {
      color: "rgb(210,210,210)", // Etiket rengini mavi yapar
    },
  }}
  sx={{
    maxWidth: "50",
    display: "block", // Bu satır eklenerek input box'ları alt alta yerleştirme sağlanır
    marginBottom: "16px", // Aralarına boşluk eklemek için marginBottom kullanabilirsiniz
    "& .MuiOutlinedInput-root": {
      "&:hover fieldset": {
        borderColor: "rgb(210,210,210)", // Hover durumunda border rengi
      },
      "&.Mui-focused fieldset": {
        borderColor: "rgb(255,0,71)", // Focus durumunda border rengi
      },
    },
  }}
/>

          ))}
          <Button
            variant="contained"
            color="success"
            onClick={generateCommand}
            sx={{ marginTop: 2, backgroundColor: "rgb(255,0,71)" }}
          >
            Komut Üret
          </Button>
        </Box>
      )}

      {/* Üretilen komutu göster */}
      {generatedCommand && (
        <Box
          sx={{
            marginTop: 2,
            padding: 2,
            backgroundColor: "#1e1e1e",
            border: "1px solid rgb(255,0,71)",
          }}
        >
          <Typography variant="h6" fontFamily={'Ubuntu Mono, Monospace'} color='rgb(255,0,71)'>Üretilen Komut</Typography>
          <Typography
            component="code"
            style={{ whiteSpace: "pre-wrap", color: "#00ff00", fontFamily:'Ubuntu Mono, Monospace', fontSize:16 }}
          >
            {generatedCommand}
          </Typography>
        </Box>
      )}
    </Box>
  );
};
